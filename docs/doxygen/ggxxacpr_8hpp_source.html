656 template<typename E>
657 struct bitflags : std::false_type {};
658 template<> struct bitflags<ActionState> : std::true_type {};
659 template<> struct bitflags<AttackState> : std::true_type {};
660 template<> struct bitflags<CommandState> : std::true_type {};
661 template<> struct bitflags<GuardState> : std::true_type {};
662 template<> struct bitflags<BackgroundState> : std::true_type {};

664 template<> struct bitflags<RawControllerInput> : std::true_type {};
665 template<> struct bitflags<GameModeFeatureFlags> : std::true_type {};
666
667 template<typename E>
668 constexpr std::enable_if_t<bitflags<E>::value, E>
669 operator|(E a, E b) noexcept {
670 using U = std::underlying_type_t<E>;
671 return static_cast<E>(
672 static_cast<U>(a) | static_cast<U>(b));
673 }
674 template <typename E>
675 constexpr std::enable_if_t<bitflags<E>::value, E>
676 operator&(E a, E b) noexcept {
677 using U = std::underlying_type_t<E>;
678 return static_cast<E>(
679 static_cast<U>(a) & static_cast<U>(b));
680 }
681 template <typename E>
682 constexpr std::enable_if_t<bitflags<E>::value, E>
683 operator~(E flags) noexcept {
684 using U = std::underlying_type_t<E>;
685 return static_cast<E>(~static_cast<U>(flags));
686 }
687 template <typename E>
688 constexpr std::enable_if_t<bitflags<E>::value, E>
689 operator|=(E& a, E b) noexcept {
690 a = a | b;
691 return a;
692 }
693 template <typename E>
694 constexpr std::enable_if_t<bitflags<E>::value, E>
695 operator&=(E& a, E b) noexcept {
696 a = a & b;
697 return a;
698 }
